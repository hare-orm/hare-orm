import httpx

from hare.contrib.frameworks.fastapi import HareFastAPI
from plugin_models import Note

application = HareFastAPI(hare_config="plugin_settings.HARE_ORM")


@application.post("/notes/{note_id}", status_code=201)
async def create_note(note_id: int) -> None:
    await Note.objects.create(id=note_id, text="written by the application")


async def write_through_application(note_id):
    async with application.router.lifespan_context(application):
        transport = httpx.ASGITransport(app=application)
        async with httpx.AsyncClient(transport=transport, base_url="http://test") as client:
            assert (await client.post(f"/notes/{note_id}")).status_code == 201
